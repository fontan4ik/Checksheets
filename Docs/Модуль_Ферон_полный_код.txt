Функция ПолучитьТаблицуЦенFeron()
	
	ТаблицаЦен = ПолучитьПустуюТаблицуЦенFeron();
	
	Сервер = СерверFeron();
	Порт = 443;
	SSL = Новый ЗащищенноеСоединениеOpenSSL;
	КлючАПИ = КлючАпиFeron();
	
	Соединение = Новый HTTPСоединение(Сервер, Порт, , , , , SSL);
	
	Запрос = Новый HTTPЗапрос; 
	
	Запрос.Заголовки.Вставить("accept", "application/json");
	Запрос.Заголовки.Вставить("Content-Type", "application/json");
	Запрос.Заголовки.Вставить("API-KEY", КлючАПИ);
	
	Запрос.АдресРесурса = "/v1/prices/list";
	
	Попытка
		
		Ответ = Соединение.ОтправитьДляОбработки(Запрос);	
	    ТелоОтвета = Ответ.ПолучитьТелоКакСтроку();
		
		Если Ответ.КодСостояния = 200 Тогда
			
			Результат = ПрочитатьСтрокуJSONВМассивСтруктур(ТелоОтвета);
			Для Каждого ДанныеПоТовару Из Результат Цикл
				
				Для Каждого ДанныеПоЦене Из ДанныеПоТовару.prices Цикл
					Если ДанныеПоЦене.type = "mic" Тогда
						НоваяСтрока = ТаблицаЦен.Добавить();
						НоваяСтрока.ИдентификаторСайта = ДанныеПоТовару.code; 
						НоваяСтрока.Цена = ДанныеПоЦене.price;	
					КонецЕсли;
				КонецЦикла;
					
			КонецЦикла;
			
		Иначе
			ЗаписатьОшибкуВЖурналРегистрации(НСтр("ru = 'Загрузка цен и остатков по поставщику ""Feron""'"), 
				СтрШаблон(НСтр("ru = 'Не удалось получить данные по ценам.
					|Ответ сервера: %1.'"), ТелоОтвета));
		КонецЕсли;
		
	Исключение
		ЗаписатьОшибкуВЖурналРегистрации(НСтр("ru = 'Загрузка цен и остатков по поставщику ""Feron""'"), 
			СтрШаблон(НСтр("ru = 'Не удалось получить данные по ценам по причине:
				|%1.'"), ПодробноеПредставлениеОшибки(ИнформацияОбОшибке())));

Процедура ОбновитьКлючСессииETM(Соединение)

	// то получаем ключ заново

		ИЛИ (ТекущаяДата() - АктуальностьКлючаСессииETM >= 60*60*2) Тогда

		Запрос.АдресРесурса = СтроковыеФункцииКлиентСервер.ВставитьПараметрыВСтроку(

		Попытка
			
			Ответ = Соединение.ОтправитьДляОбработки(Запрос);
			ТелоОтвета = Ответ.ПолучитьТелоКакСтроку();
			 
			Если Ответ.КодСостояния = 200 Тогда
				
				Результат = ПрочитатьСтрокуJSONВМассивСтруктур(ТелоОтвета);
				Если НЕ ПустаяСтрока(Результат.data.session) Тогда
					КлючСессииETM = Результат.data.session;
					АктуальностьКлючаСессииETM = ТекущаяДата();
				Иначе
